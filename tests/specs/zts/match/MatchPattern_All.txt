== should format a variant pattern with bindings ==
const a = match (s) { Circle{radius}=>radius };

[expect]
const a = match (s) {
    Circle { radius } => radius,
};

== should format a variant pattern that binds nothing ==
const a = match (s) { Circle{}=>1,Square{}=>2 };

[expect]
const a = match (s) {
    Circle {} => 1,
    Square {} => 2,
};

== should format a renamed binding ==
const a = match (s) { Circle{radius:r}=>r };

[expect]
const a = match (s) {
    Circle { radius: r } => r,
};

== should format a binding with a default ==
const a = match (s) { Circle { radius = 1 } => radius };

[expect]
const a = match (s) {
    Circle { radius = 1 } => radius,
};

== should format a rest binding ==
const a = match (s) { Circle { radius, ...rest } => radius };

[expect]
const a = match (s) {
    Circle { radius, ...rest } => radius,
};

== should format string literal patterns ==
const a = match (l) { 'debug'=>0,'info'=>1 };

[expect]
const a = match (l) {
    "debug" => 0,
    "info" => 1,
};
