-- file.zts --
== should format ==
union DeleteOutcome='soft'|'hard'|'unknown';

[expect]
union DeleteOutcome = "soft" | "hard" | "unknown";

== should drop a leading bar on the single line form ==
union Level = | 'info' | 'warn';

[expect]
union Level = "info" | "warn";

== should format when exported ==
export union Level = 'info' | 'warn' | 'error';

[expect]
export union Level = "info" | "warn" | "error";

== should keep a multi-line union multi-line ==
union Level =
  | 'info'
  | 'warn'
  | 'error';

[expect]
union Level =
    | "info"
    | "warn"
    | "error";

== should break a long union on to multiple lines ==
union HttpMethod = 'get' | 'put' | 'post' | 'patch' | 'delete' | 'options' | 'head' | 'trace' | 'connect' | 'query' | 'purge';

[expect]
union HttpMethod =
    | "get"
    | "put"
    | "post"
    | "patch"
    | "delete"
    | "options"
    | "head"
    | "trace"
    | "connect"
    | "query"
    | "purge";

== should format a single member ==
union Only = 'one';

[expect]
union Only = "one";

== should format numeric members ==
union HttpStatus=200|404|500;

[expect]
union HttpStatus = 200 | 404 | 500;

== should format negative numeric members ==
union Offset = -1|0|1;

[expect]
union Offset = -1 | 0 | 1;

== should format mixed string and numeric members ==
union Ans = 42|'unknown';

[expect]
union Ans = 42 | "unknown";

== should keep the raw form of a numeric member ==
union Mask = 0x10|0o17|1_000;

[expect]
union Mask = 0x10 | 0o17 | 1_000;

== should break a long numeric union on to multiple lines ==
union Port = 3000 | 3001 | 3002 | 3003 | 3004 | 3005 | 3006 | 3007 | 3008 | 3009 | 3010 | 3011 | 3012 | 3013 | 3014 | 3015 | 3016 | 3017;
