== should keep a comment above an enum ==
// the shapes
enum Shape {
  Circle { r: number },
}

[expect]
// the shapes
enum Shape {
    Circle { r: number },
}

== should keep comments on enum variants ==
enum Shape {
  // a round one
  Circle { r: number },
  // a square one
  Square { s: number },
}

[expect]
enum Shape {
    // a round one
    Circle { r: number },
    // a square one
    Square { s: number },
}

== should keep a comment on an enum field ==
enum Shape {
  Circle {
    // in pixels
    r: number,
  },
}

[expect]
enum Shape {
    Circle {
        // in pixels
        r: number,
    },
}

== should keep a trailing comment on a variant ==

}

[expect]

    Circle { r: number }, // round
    Square { s: number }, // square
}

};

    Circle { r } => r, // round
    _ => 0, // fallback
};
